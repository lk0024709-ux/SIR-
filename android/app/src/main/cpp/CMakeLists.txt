cmake_minimum_required(VERSION 3.22.1)
project(sir_android_native LANGUAGES C CXX)

if(NOT ANDROID)
    message(FATAL_ERROR "This CMake project is intended to be built by the Android Gradle plugin/NDK.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Mobile CPU-only build policy. Reuse the one repository-pinned llama.cpp FetchContent
# integration; do not vendor or fetch a second llama.cpp copy here.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build llama.cpp as static libraries for the JNI shared object" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_HIP OFF CACHE BOOL "" FORCE)
set(GGML_MUSA OFF CACHE BOOL "" FORCE)

get_filename_component(SIR_REPOSITORY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(SIR_LLAMA_CMAKE "${SIR_REPOSITORY_ROOT}/cmake/llama_cpp.cmake")
if(NOT EXISTS "${SIR_LLAMA_CMAKE}")
    message(FATAL_ERROR "Expected the existing pinned SIR llama.cpp integration at ${SIR_LLAMA_CMAKE}")
endif()
include("${SIR_LLAMA_CMAKE}")

add_library(sir_android SHARED native-lib.cpp)
find_library(ANDROID_LOG_LIBRARY log REQUIRED)
target_link_libraries(sir_android PRIVATE llama ${ANDROID_LOG_LIBRARY})
